import fs from 'node:fs';
import http from 'node:http';
import {DecisionBridge} from '/Volumes/Data/symphony-ts/packages/decision/src/index.ts';
const dir='/private/tmp/symphony-pr159-live-20261023';
const authToken=fs.readFileSync('/private/tmp/symphony-pr156-reacceptance-token','utf8').trim();
const log=(entry:unknown)=>fs.appendFileSync(dir+'/requests.ndjson',JSON.stringify({at:new Date().toISOString(),...entry as object})+'\n');
const bridge=new DecisionBridge({storeDir:dir+'/store',host:'127.0.0.1',port:4041,authToken,allowedOrigins:['https://chatgpt.com']});
await bridge.start();
const proxy=http.createServer(async(req,res)=>{
 const chunks:Buffer[]=[];for await(const chunk of req)chunks.push(Buffer.from(chunk));const body=Buffer.concat(chunks);
 let progress:unknown;try{const parsed=JSON.parse(body.toString());if(req.url?.endsWith('/heartbeat'))progress=parsed.progress;}catch{}
 const control=JSON.parse(fs.readFileSync(dir+'/control.json','utf8'));
 if(req.method==='POST'&&req.url?.endsWith('/result')&&control.holdResult){log({method:req.method,path:req.url,action:'hold_result'});while(JSON.parse(fs.readFileSync(dir+'/control.json','utf8')).holdResult){if(req.socket.destroyed)return;await new Promise(r=>setTimeout(r,100));}}
 const upstream=http.request({hostname:'127.0.0.1',port:4041,path:req.url,method:req.method,headers:req.headers},up=>{log({method:req.method,path:req.url,status:up.statusCode,...progress?{progress}:{}});
 if(req.method==='POST'&&req.url?.endsWith('/result')&&control.dropAckOnce&&up.statusCode===200){up.resume();up.on('end',()=>{const next=JSON.parse(fs.readFileSync(dir+'/control.json','utf8'));next.dropAckOnce=false;fs.writeFileSync(dir+'/control.json',JSON.stringify(next));log({method:req.method,path:req.url,action:'accepted_ack_dropped'});res.destroy();});return;}
 res.writeHead(up.statusCode??502,up.headers);up.pipe(res);});
 upstream.on('error',()=>{if(!res.headersSent)res.writeHead(502);res.end();});upstream.end(body);
});
await new Promise<void>(resolve=>proxy.listen(4040,'127.0.0.1',resolve));
console.log('Isolated production Decision Bridge ready');
const stop=async()=>{await new Promise<void>(r=>proxy.close(()=>r()));await bridge.stop();console.log('Bridge stopped');process.exit(0);};process.on('SIGINT',stop);process.on('SIGTERM',stop);
