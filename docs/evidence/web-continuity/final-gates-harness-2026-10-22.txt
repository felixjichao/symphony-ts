import fs from 'node:fs';
import http from 'node:http';
import {execFileSync} from 'node:child_process';
import {DecisionBridge} from '/Volumes/Data/symphony-ts/packages/decision/src/index.ts';
import {createHost} from '/Volumes/Data/symphony-ts/apps/cli/src/host.ts';
import {authorizeDecisionWebExecution} from '/Volumes/Data/symphony-ts/apps/cli/src/decision-web-admission.ts';
import {createGitHubAdapterProfile} from '/Volumes/Data/symphony-ts/packages/tracker/src/github/profile.ts';
import {createStructuredLogger} from '/Volumes/Data/symphony-ts/packages/observability/src/index.ts';
const dir='/private/tmp/symphony-pr156-final-gates-20261022';
process.env.GITHUB_TOKEN=execFileSync('gh',['auth','token'],{encoding:'utf8'}).trim();
const authToken=fs.readFileSync('/private/tmp/symphony-pr156-reacceptance-token','utf8').trim();
const log=(file:string,entry:any)=>fs.appendFileSync(dir+'/'+file,JSON.stringify({at:new Date().toISOString(),...entry})+'\n');
const bridge=new DecisionBridge({storeDir:dir+'/store',host:'127.0.0.1',port:4041,authToken,allowedOrigins:['https://chatgpt.com'],authorizeExecution:async(task)=>{
 const allowed=await authorizeDecisionWebExecution(task);
 if(task.kind==='review')log('admission.ndjson',{taskId:task.id,target:task.target,allowed});
 return allowed;
}});
await bridge.start();
const proxy=http.createServer(async(req,res)=>{
 if(req.method==='GET'&&req.url?.endsWith('/context')){
  while(JSON.parse(fs.readFileSync(dir+'/control.json','utf8')).holdContext){
   if(req.socket.destroyed)return;
   await new Promise(r=>setTimeout(r,100));
  }
 }
 const upstream=http.request({hostname:'127.0.0.1',port:4041,path:req.url,method:req.method,headers:req.headers},up=>{
  log('requests.ndjson',{method:req.method,path:req.url,status:up.statusCode});res.writeHead(up.statusCode??502,up.headers);up.pipe(res);
 });upstream.on('error',()=>{if(!res.headersSent)res.writeHead(502);res.end();});req.pipe(upstream);
});
await new Promise<void>(resolve=>proxy.listen(4040,'127.0.0.1',resolve));
const profile=createGitHubAdapterProfile();
const scopedProfile={...profile,createAdapter(context:any){const adapter=profile.createAdapter(context);return {kind:adapter.kind,
 fetchIssuesByStates:async(states:any)=>(await adapter.fetchIssuesByStates(states)).filter(i=>i.nativeRef?.repo==='felixjichao/symphony-delivery-dogfood'&&i.nativeRef.number===24),
 fetchIssuesByIds:async(ids:any)=>(await adapter.fetchIssuesByIds(ids)).filter(i=>i.nativeRef?.repo==='felixjichao/symphony-delivery-dogfood'&&i.nativeRef.number===24)};}};
const host=await createHost({workflowPath:dir+'/WORKFLOW.md',trackerProfiles:[scopedProfile],decisionBridgeUrl:'http://127.0.0.1:4040',decisionBridgeToken:authToken,
 canDispatch:()=>false,logger:createStructuredLogger({sinks:[{write(line:string){try{const entry=JSON.parse(line);log('host-events.ndjson',{event:entry.event,outcome:entry.outcome,reason:entry.reason});}catch{}}}]})});
await host.start();
let last='';const timer=setInterval(()=>{
 const observations=host.workContinuity.observations.map(o=>({workKey:o.workKey,result:o.result,diagnosis:o.diagnosis}));
 const snapshot={runningWorkers:host.state.running.size,observations};const key=JSON.stringify(snapshot);
 if(key!==last){last=key;log('host-observations.ndjson',snapshot);}
},500);
console.log('Real host native poll + GitHub adapter scoped to dogfood #24, production GitHub review admission, isolated Bridge ready; coding denied');
let stopping=false;const stop=async()=>{if(stopping)return;stopping=true;clearInterval(timer);await host.stop();await new Promise<void>(resolve=>proxy.close(()=>resolve()));await bridge.stop();console.log('Host and Bridge stopped');process.exit(0);};
process.on('SIGTERM',stop);process.on('SIGINT',stop);
